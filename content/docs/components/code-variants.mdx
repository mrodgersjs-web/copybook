## Usage

Render each variant you show on the server, then pass them in. Mix in a prompt with `kind: "prompt"` — the same pattern as [Command Tabs](/docs/components/command-tabs) with pnpm, npm, yarn, bun, and a prompt:

```tsx
import { listVariants, render } from "@copybook/core";

import { CodeVariants } from "@/components/copybook/code-variants";
import { tryIt } from "@/content/ai-gateway";
import { tryItPrompt } from "@/content/try-it";

export const TryIt = ({ endpoint, model }) => (
  <CodeVariants
    defaultView="rendered"
    items={[
      {
        key: "prompt",
        kind: "prompt",
        label: "Prompt",
        rendered: render(tryItPrompt, { endpoint, model }),
      },
      ...listVariants(tryIt)
        .filter((variant) => variant.when.scenario === "chat")
        .map((variant) => ({
          key: variant.key,
          label: variant.label,
          rendered: render(
            tryIt,
            { endpoint, model },
            { variant: variant.key }
          ),
        })),
    ]}
    viewToggle
  />
);
```

Prompt tabs can format their Markdown, like [Agent Prompt's rendered view](/docs/components/agent-prompt#rendered-view). Set `defaultView="rendered"`, and add `viewToggle` for a Preview / Markdown switch. The copy button still copies the Markdown.

## Props

| Prop | Type | Description |
| --- | --- | --- |
| `items` | `{ key, label, rendered, kind? }[]` | One tab each, in order. `kind: "prompt"` wraps long lines and adds an agent icon. |
| `defaultKey` | `string` | The tab that starts selected. |
| `defaultView` | `"source" \| "rendered"` | The view prompt tabs show first. Defaults to `"source"`. |
| `viewToggle` | `boolean` | Shows a Preview / Markdown switch on prompt tabs. |
| `onCopy` | `(event: CopyEvent) => void` | Called with `{ id, variant, text }` after a copy. |
